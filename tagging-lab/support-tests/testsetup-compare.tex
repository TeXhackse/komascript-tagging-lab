\NewDocumentCommand{\addpart}{O{#2}m}{
  \edef\reserved@a{%
    \unexpanded{%
      \part[{#1}]{#2}%
      \c@secnumdepth=
    }\the\c@secnumdepth\relax
  }%
  \c@secnumdepth=\numexpr \UseName{toclevel@part}-1\relax
  \reserved@a
  \let\@elt\@stpelt
  \csname cl@part\endcsname
}
